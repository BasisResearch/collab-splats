```python
"""Smoke test: bae 0.2.3 pre-installed at /opt/bae (editable, USE_CUDSS=1)."""
import pytest
import sys
import types
import importlib.util
import pathlib


def test_bae_imports():
    import pypose  # must import before bae (pypose.autograd accesses bae at load time)
    import bae
    from bae.autograd.function import TrackingTensor, map_transform
    from bae.utils.pysolvers import PCG
    from bae.optim import LM
    assert bae.__file__ is not None


def test_bae_cuda_version():
    import torch
    assert "12.1" in torch.version.cuda, f"Expected CUDA 12.1, got {torch.version.cuda}"
    assert torch.__version__.startswith("2.4"), f"Expected torch 2.4.x, got {torch.__version__}"


def test_bundle_adjustment_module_loads():
    """Load bundle_adjustment directly, bypassing collab_splats __init__ which
    auto-imports MapAnythingCreator and can fail if feedforward env is broken."""
    import pypose  # must import first

    for pkg in ("collab_splats", "collab_splats.pointcloud"):
        if pkg not in sys.modules:
            sys.modules[pkg] = types.ModuleType(pkg)

    ba_path = pathlib.Path("/workspace/collab-splats/collab_splats/pointcloud/bundle_adjustment.py")
    spec = importlib.util.spec_from_file_location(
        "collab_splats.pointcloud.bundle_adjustment", ba_path
    )
    mod = importlib.util.module_from_spec(spec)
    sys.modules["collab_splats.pointcloud.bundle_adjustment"] = mod
    spec.loader.exec_module(mod)

    assert hasattr(mod, "run_bundle_adjustment")
    assert hasattr(mod, "extract_tracks_vggsfm")
```

## Task 7: Create setup_nerfstudio.sh

Installs local nerfstudio from `/workspace/nerfstudio/` after patching 5 version conflicts in its pyproject.toml. Reinstalls gsplat-rade last to prevent nerfstudio from overwriting the fork with official gsplat 1.4.0.

**Files:**
- Create: `setup_nerfstudio.sh`

- [ ] **Step 1: Write the script**

```bash
#!/bin/bash
# Install local nerfstudio with cu121 compatibility patches.
#
# Patches applied to /workspace/nerfstudio/pyproject.toml:
#   1. gsplat==1.4.0        → gsplat-rade fork (prevents overwriting our RADE fork)
#   2. timm==0.6.7          → timm>=0.6.7 (uniception requires >=0.9)
#   3. viser==1.0.0         → viser>=0.2.0 (feedforward uses >=0.2.23)
#   4. opencv-python-headless removed (collab-splats provides opencv-python, a superset)
#   5. nerfacc==0.5.2       → nerfacc>=0.5.2 (exact pin may fail with torch 2.4)
#
# gsplat-rade is reinstalled last so nerfstudio's dep resolver cannot overwrite it.
#
# TODO: hloc install deferred — pycolmap 4.0.4 vs hloc compatibility unresolved.
#   hloc v1.4 requires pycolmap ~0.4. Validate which hloc commit supports pycolmap 4.x
#   before adding install step here.

set -e

NS_DIR="/workspace/nerfstudio"
PYTHON="/opt/conda/envs/nerfstudio/bin/python"
PIP="/opt/conda/envs/nerfstudio/bin/pip"

echo "=== setup_nerfstudio: patching nerfstudio/pyproject.toml ==="

$PYTHON - <<'PYEOF'
import re, pathlib

p = pathlib.Path("/workspace/nerfstudio/pyproject.toml")
text = p.read_text()

patches = [
    # gsplat exact pin → rade fork
    ('"gsplat==1.4.0"',
     '"gsplat @ git+https://github.com/brian-xu/gsplat-rade.git"'),
    # timm exact pin → lower-bound only
    ('"timm==0.6.7"',
     '"timm>=0.6.7"'),
    # viser exact pin → lower-bound only
    ('"viser==1.0.0"',
     '"viser>=0.2.0"'),
    # opencv-python-headless → remove (collab-splats provides opencv-python, a superset)
    ('"opencv-python-headless==4.10.0.84",\n',
     ''),
    # nerfacc exact pin → lower-bound only
    ('"nerfacc==0.5.2"',
     '"nerfacc>=0.5.2"'),
]

original = text
for old, new in patches:
    if old not in text:
        print(f"  WARNING: pattern not found, skipping: {old!r}")
        continue
    text = text.replace(old, new, 1)
    print(f"  patched: {old!r} → {new!r}")

if text == original:
    print("  No changes made (already patched?)")
else:
    p.write_text(text)
    print("  nerfstudio/pyproject.toml written.")
PYEOF

echo "=== setup_nerfstudio: installing local nerfstudio ==="
$PIP install --no-cache-dir -e "$NS_DIR"

echo "=== setup_nerfstudio: reinstalling gsplat-rade (must win over nerfstudio's official gsplat) ==="
$PIP install --no-cache-dir --no-build-isolation \
    git+https://github.com/brian-xu/gsplat-rade.git

echo "=== setup_nerfstudio: smoke test ==="
$PYTHON - <<'PYEOF'
import nerfstudio, gsplat, torch
gsplat_path = gsplat.__file__
assert "gsplat" in gsplat_path
print(f"[OK] nerfstudio={nerfstudio.__version__ if hasattr(nerfstudio,'__version__') else 'installed'}")
print(f"[OK] gsplat path: {gsplat_path}")
print(f"[OK] torch={torch.__version__}  cuda={torch.version.cuda}")
# Verify gsplat-rade specific symbol exists (not in official gsplat)
try:
    from gsplat import rasterization_2dgs_inria_wrapper  # rade-specific
    print("[OK] gsplat-rade fork confirmed (rasterization_2dgs_inria_wrapper present)")
except ImportError:
    print("[WARN] gsplat-rade specific symbol not found — may have official gsplat installed")
PYEOF

echo ""
echo "=== setup_nerfstudio complete ==="
echo "NOTE: hloc not installed. See TODO in this script for pycolmap 4.x compat details."
```
