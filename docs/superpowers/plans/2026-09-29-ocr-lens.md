> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a query-free LLaVA-1.6 OCR-lens extractor (`semantics.extractor: ocr_lens`) whose per-point codes decode to English words, and reorder the semantics stage to compress per frame before lifting so no stage holds all frames or a full-width `(P, D)`.

**Architecture:** The 2D patch cache becomes fp16 and is keyed on extractor kwargs. `FeatureAutoencoder.fit` streams frames out of that zarr cache. `lift_features` takes a per-frame loader, so the reconstructor lifts AE codes frame by frame. `ocr_lens.py` holds one registered extractor class and four plain functions: `word_vocabulary`, `verbalize`, `load_decoder` and `score_ocr_heads`.

**Tech Stack:** torch, transformers 4.57.6 (`LlavaNextForConditionalGeneration`), zarr 3, nltk WordNet, safetensors, pyarrow, PIL.

| spec says | plan does | why |
|---|---|---|
| reconstructor keeps `zarr_path.exists()` skip | new `valid_feature_cache(...)` checked before the extractor is built | `exists()` would reuse a cache from another `layer`. Building the extractor to ask `extract_feature_cache` would load 7B weights on every cache hit. |

| GPU parity as a pytest | parity is Gate 0, a scratch script against the prototype's cached `states_llava16_L17.pt` | `ref_image.jpg` and the prototype are not in the repo. |

```python
def valid_feature_cache(
    cache_dir: Path, name: str, images_dir: Path, extractor_kwargs: Optional[dict[str, Any]] = None
) -> Optional[Path]:
    """
    Path of a reusable `<name>.zarr` patch cache, or None when it must be (re)extracted.

    - valid: extractor name, frame count and extractor kwargs all match the store's attrs
    - needs no extractor instance, so a cache hit never loads the model

    Args:
        cache_dir: directory holding the 2D patch caches.
        name: extractor registry name.
        images_dir: the scene's images/ directory; its frame count must match.
        extractor_kwargs: constructor kwargs the cache was extracted with; None means none.

    Returns:
        The store path, or None.
    """
    zarr_path = Path(cache_dir) / f"{name}.zarr"
    n_frames = len(frame_paths(images_dir))
    expected = {"extractor": name, "n_frames": n_frames, "extractor_kwargs": extractor_kwargs or {}}
    if open_valid(zarr_path, expected) is None:
        return None
    return zarr_path


def extract_feature_cache(
    extractor: BaseFeatureExtractor,
    images_dir: Path,
    cache_dir: Path,
    extractor_kwargs: Optional[dict[str, Any]] = None,
) -> Path:
    """
    Extract patch features from a scene's images/ into `cache_dir/<extractor>.zarr`.

    - re-entrant: a store `valid_feature_cache` accepts is returned untouched
    - the attrs that make a store look valid are written LAST, after every frame is on disk
    - so a run that dies mid-extraction leaves an attr-less store the next run re-extracts
    - stored as float16: half the disk, and consumers cast to float32 on read
    - the check ignores resolution and resize mode; delete the store after changing either

    Args:
        extractor: supplies `.name`, `.patch_size` and `.forward`.
        images_dir: the scene's images/ directory of frame_NNNNNN.<ext> keyframes.
        cache_dir: directory to write the 2D patch cache into.
        extractor_kwargs: constructor kwargs `extractor` was built with; they key the cache.

    Returns:
        Path of the store; `features` is (N, D, H_p, W_p) float16, one chunk per frame.

    Raises:
        FileNotFoundError: when `images_dir` holds no frame images.
    """
    zarr_path = Path(cache_dir) / f"{extractor.name}.zarr"
    extractor_kwargs = extractor_kwargs or {}

    # Decode one path at a time: RAM holds one frame, not the stack
    paths = frame_paths(images_dir)
    if not paths:
        raise FileNotFoundError(f"No frame images ({list(IMAGE_EXTS)}) in {images_dir}")
    N = len(paths)

    # A cache is valid when the extractor name, frame count and kwargs all match
    if valid_feature_cache(cache_dir, extractor.name, images_dir, extractor_kwargs) is not None:
        logger.info("Feature cache valid, skipping extraction: %s", zarr_path)
        return zarr_path

    Path(cache_dir).mkdir(parents=True, exist_ok=True)

    # Probe the first frame to learn (D, H_p, W_p) before allocating the store
    first_frame = read_image(paths[0])
    with torch.no_grad():
        [first_feat] = extractor.forward([first_frame])
    D, H_p, W_p = first_feat.shape

    store = zarr.open(str(zarr_path), mode="w")

    # One chunk per frame: reading frame i loads exactly 1 disk chunk
    arr = store.create_array(
        "features",
        shape=(N, D, H_p, W_p),
        chunks=(1, D, H_p, W_p),
        dtype="float16",
        fill_value=0,
    )
    arr[0] = first_feat.cpu().half().numpy()

    # Iterate the rest lazily — never more than one frame in RAM
    for i in range(1, N):
        frame = read_image(paths[i])
        with torch.no_grad():
            [feat] = extractor.forward([frame])
        arr[i] = feat.cpu().half().numpy()
        logger.debug("extract_feature_cache: %d/%d frames written", i + 1, N)

    # Validity attrs last: a crash mid-loop leaves a store the check above rejects
    store.attrs.update(
        {
            "extractor": extractor.name,
            "patch_size": extractor.patch_size,
            "n_frames": N,
            "extractor_kwargs": extractor_kwargs,
        }
    )

    logger.info("Feature cache written: %s  shape=%s", zarr_path, tuple(arr.shape))
    return zarr_path
```

```python
def _sample_at_source_pixels(
    frame_features: Callable[[int], torch.Tensor],
    pixel_indices: np.ndarray,
    image_size: tuple[int, int],
    dim: int,
    device: torch.device,
) -> torch.Tensor:
    """
    Source-frame-only lift: each point sampled at its own (frame_id, row, col).

    - fallback for points with zero multi-view weight
    - zero weight: never visible, or always depth-inconsistent
    - loads only the frames that hold such points
    """
    P = len(pixel_indices)
    H, W = image_size
    out = torch.zeros((P, dim), dtype=torch.float32, device=device)

    for i in np.unique(pixel_indices[:, 0]):
        mask_i = pixel_indices[:, 0] == i
        fmap = frame_features(int(i))
        fmap = fmap.to(device=device, dtype=torch.float32)

        # Clip to model frame to handle any rounding drift from upstream
        rows = np.clip(pixel_indices[mask_i, 1], 0, H - 1)
        cols = np.clip(pixel_indices[mask_i, 2], 0, W - 1)
        rows_f = rows.astype(np.float32)
        cols_f = cols.astype(np.float32)
        rows_t = torch.from_numpy(rows_f)
        cols_t = torch.from_numpy(cols_f)
        rows_t = rows_t.to(device)
        cols_t = cols_t.to(device)

        sampled = _grid_sample_at_pixels(fmap, rows_t, cols_t, image_size)
        mask_t = torch.from_numpy(mask_i)
        out[mask_t.to(device)] = sampled
    return out
```

```python
def _stream_frames(features: zarr.Array, device: torch.device) -> Iterator[Tensor]:
    """
    Frames of a patch cache as (H_p * W_p, D) float32 on device, shuffled, one read ahead.

    - the next frame's chunk is read on a worker thread while the current one trains
    - RAM holds at most the current and the prefetched frame
    """
    order = torch.randperm(features.shape[0]).tolist()

    with ThreadPoolExecutor(max_workers=1) as pool:
        pending = pool.submit(features.__getitem__, order[0])

        for k in range(len(order)):
            frame = pending.result()
            if k + 1 < len(order):
                pending = pool.submit(features.__getitem__, order[k + 1])

            patches = torch.from_numpy(frame)
            patches = patches.to(device=device, dtype=torch.float32)
            yield patches.flatten(1).T


def _batches(features: Union[Tensor, zarr.Array], batch_size: int, device: torch.device) -> Iterator[Tensor]:
    """
    One epoch of shuffled (B, D) float32 training batches.

    - tensor: one permutation over all rows
    - zarr patch cache: frames in shuffled order, patches shuffled within each frame
    """
    if isinstance(features, Tensor):
        perm = torch.randperm(features.shape[0], device=features.device)
        for start in range(0, len(perm), batch_size):
            yield features[perm[start : start + batch_size]]
        return

    for patches in _stream_frames(features, device):
        perm = torch.randperm(patches.shape[0], device=device)
        for start in range(0, len(perm), batch_size):
            yield patches[perm[start : start + batch_size]]
```

```python
    Shortest WordNet base form over noun / verb / adjective readings; None if not a WordNet word.

    - trees -> tree, leaves -> leaf; drops fragments ("gry") and function words ("the")
    """
    forms = [form for pos in (wn.NOUN, wn.VERB, wn.ADJ) if (form := wn.morphy(word, pos))]
    if not forms:
        return None
    return min(forms, key=len)


def word_vocabulary(tokenizer: PreTrainedTokenizerBase, words: Optional[list[str]] = None) -> WordVocab:
    """
    Decode vocabulary over the tokenizer's word-initial (`▁`) ASCII-letter tokens.

    - words None: every token whose WordNet base form exists, merged by lemma (~4.8k for Llama-2)
    - words given: each word collects every token whose lemma or lowercase text matches it
    - words with no single-token form are dropped with one warning

    Args:
        tokenizer: the model tokenizer; only `len` and `convert_ids_to_tokens` are used.
        words: custom label set; None for the default English vocabulary.

    Returns:
        The WordVocab, words in first-seen token order (default) or the given order (custom).

    Raises:
        LookupError: when the WordNet corpus is not installed.
    """
    # Fail with the fix, not nltk's generic message
    try:
        wn.ensure_loaded()
    except LookupError as e:
        raise LookupError("WordNet corpus missing: python -m nltk.downloader wordnet") from e

    # Word-initial ASCII tokens with their lemma and lowercase text
    pattern = re.compile(r"^▁([A-Za-z]{2,})$")
    keyed = []
    for token_id in range(len(tokenizer)):
        match = pattern.match(tokenizer.convert_ids_to_tokens(token_id))
        if match is None:
            continue
        text = match.group(1).lower()
        keyed.append((token_id, _lemma(text), text))

    # Default: every WordNet lemma is a word
    if words is None:
        pairs = [(token_id, lemma) for token_id, lemma, _ in keyed if lemma is not None]
        labels = list(dict.fromkeys(lemma for _, lemma in pairs))
    else:
        # Custom: a token joins the word whose lemma or text equals its own lemma or text
        targets = {}
        for word in words:
            text = word.lower()
            targets[text] = word
            targets[_lemma(text) or text] = word

        pairs = []
        for token_id, lemma, text in keyed:
            word = targets.get(lemma) or targets.get(text)
            if word is not None:
                pairs.append((token_id, word))

        # Words no token reached are dropped, once, loudly
        found = {word for _, word in pairs}
        missing = [word for word in words if word not in found]
        if missing:
            logger.warning("word_vocabulary: no single-token form for %s, dropped", missing)
        labels = [word for word in words if word in found]

    row_of = {word: row for row, word in enumerate(labels)}
    token_ids = torch.tensor([token_id for token_id, _ in pairs], dtype=torch.long)
    word_index = torch.tensor([row_of[word] for _, word in pairs], dtype=torch.long)
    return WordVocab(words=labels, token_ids=token_ids, word_index=word_index)
```

```python
########################################################################
# Decoding
########################################################################


@torch.no_grad()
def verbalize(
    features: torch.Tensor, decoder: nn.Module, vocab: WordVocab, *, k: int = 10, chunk: int = 256
) -> tuple[list[list[str]], np.ndarray, np.ndarray]:
    """
    Top-k vocabulary words per lens state; query-free counterpart of `score_queries`.

    - decoder softmax over the full token vocabulary, token probs summed per word, renormalized
    - a (D, H, W) map is flattened row-major to (H * W, D) first

    Args:
        features: (P, D) lens states or decoded point codes, or a (D, H, W) map.
        decoder: final norm + lm_head, from `OCRLensExtractor.decoder` or `load_decoder`.
        vocab: from `word_vocabulary`.
        k: words per row, capped at the vocabulary size.
        chunk: rows per softmax block.

    Returns:
        (words, probs, vocab_mass): P lists of k words, (P, k) probabilities, and (P,) full-vocab
        probability landing on the vocabulary (low: the lens has no vocabulary word for it).
    """
    if features.ndim == 3:
        features = features.flatten(1).T

    param = next(decoder.parameters())
    token_ids = vocab.token_ids.to(param.device)
    word_index = vocab.word_index.to(param.device)
    k = min(k, len(vocab.words))

    words, probs, mass = [], [], []
    for start in range(0, len(features), chunk):
        x = features[start : start + chunk]
        x = x.to(device=param.device, dtype=param.dtype)
        logits = decoder(x)
        p = logits.float().softmax(-1)

        # Sum token probabilities into words, then renormalize over the vocabulary
        per_word = torch.zeros(len(p), len(vocab.words), device=p.device)
        per_word.index_add_(1, word_index, p[:, token_ids])
        in_vocab = per_word.sum(1, keepdim=True)
        normalized = per_word / in_vocab
        top_p, top_i = normalized.topk(k, dim=-1)

        words += [[vocab.words[j] for j in row] for row in top_i.tolist()]
        probs.append(top_p.cpu().numpy())
        mass.append(in_vocab[:, 0].cpu().numpy())

    return words, np.concatenate(probs), np.concatenate(mass)
```

```python
########################################################################
# Lens
########################################################################


@torch.no_grad()
def _build_lens(model: LlavaNextForConditionalGeneration, head_scores: torch.Tensor, n_heads: int) -> torch.Tensor:
    """
    Sum of W_O W_V over the top-scoring heads: the (D, D) verbalization lens, float32 on CPU.

    - GQA-aware: each kv head serves n_heads / n_kv_heads query heads (Vicuna has no GQA)
    - accumulated in float32 on CPU, before truncation, so heads above `layer` count
    """
    cfg = model.config.text_config
    layers = model.model.language_model.layers
    n_query = cfg.num_attention_heads
    head_dim = getattr(cfg, "head_dim", None) or cfg.hidden_size // n_query
    n_rep = n_query // cfg.num_key_value_heads
    hidden = cfg.hidden_size

    # Scores must index (layer, head) of this model
    if tuple(head_scores.shape) != (len(layers), n_query):
        raise ValueError(f"head_scores must be {(len(layers), n_query)}, got {tuple(head_scores.shape)}")

    lens = torch.zeros((hidden, hidden), dtype=torch.float32)
    top = torch.topk(head_scores.reshape(-1), n_heads).indices
    for index in top.tolist():
        layer, head = divmod(index, n_query)
        attn = layers[layer].self_attn
        w_o = attn.o_proj.weight[:, head * head_dim : (head + 1) * head_dim]
        w_v = attn.v_proj.weight.view(-1, head_dim, hidden)[head // n_rep]
        lens += w_o.float() @ w_v.float()
    return lens


def _anyres_grid(processor: LlavaNextProcessor, inputs: dict[str, torch.Tensor]) -> tuple[int, int]:
    """
    (rows, cols) of LLaVA-1.6's unpadded high-res token grid for one processed image.

    - uses the processor's own token-count math (private `_get_unpadded_features`, transformers 4.57)
    """
    orig_h, orig_w = inputs["image_sizes"][0].tolist()
    tile_h, tile_w = inputs["pixel_values"].shape[-2:]
    pinpoints = processor.image_processor.image_grid_pinpoints
    best_h, best_w = select_best_resolution([orig_h, orig_w], pinpoints)
    n_high_res, rows = processor._get_unpadded_features(
        orig_h,
        orig_w,
        tile_h // processor.patch_size,
        tile_w // processor.patch_size,
        best_h // tile_h,
        best_w // tile_w,
    )
    return rows, n_high_res // rows

@BaseFeatureExtractor.register("ocr_lens")
class OCRLensExtractor(BaseFeatureExtractor):
    """
    LLaVA-1.6 image-token states through the OCR-head verbalization lens, per token-grid cell.

    - language model truncated after `layer`; that layer's output is read with a forward hook
    - maps are rms_normalize(lens @ h): decoding is linear after it, so averaged states decode
      as averaged per-view logits
    - `decoder` and `vocab` turn states (or decoded point codes) into words via `verbalize`

    Args:
        model_id: LLaVA-1.6 hub id or local directory.
        layer: decoder layer whose output is read, 0-based.
        n_heads: number of top-scoring OCR heads summed into the lens.
        head_scores: (layers, heads) mean OCR scores, a path to a saved one, or None for the
            packaged LLaVA-1.6-Vicuna-7B scores.
        dtype: torch dtype name for the model weights.

    Raises:
        ValueError: when `layer` is outside the model's decoder layers, or `head_scores` does not
            match its (layers, heads).
    """

    # AnyRes has no fixed pixel patch; nominal, recorded in the cache attrs only
    patch_size = 14

    def __init__(
        self,
        model_id: str = "llava-hf/llava-v1.6-vicuna-7b-hf",
        layer: int = 17,
        n_heads: int = 102,
        head_scores: Union[torch.Tensor, str, Path, None] = None,
        dtype: str = "float16",
    ) -> None:
        super().__init__(resize_mode="max_size", image_resolution=672)
        self._device = get_device()
        self.model = LlavaNextForConditionalGeneration.from_pretrained(model_id, dtype=getattr(torch, dtype))
        self.processor = AutoProcessor.from_pretrained(model_id)
        lm = self.model.model.language_model

        # The hook reads this layer's output, so it must exist
        n_layers = len(lm.layers)
        if not 0 <= layer < n_layers:
            raise ValueError(f"layer must be in [0, {n_layers}), got {layer}")
        self.layer = layer

        # Lens from the full model's heads, then drop every layer after `layer`
        if head_scores is None:
            head_scores = Path(__file__).parent / "assets" / "llava16_vicuna7b_ocr_head_scores.pt"
        if not isinstance(head_scores, torch.Tensor):
            head_scores = torch.load(head_scores, weights_only=True)
        lens = _build_lens(self.model, head_scores, n_heads)
        lm.layers = lm.layers[: layer + 1]

        self.model.to(self._device).eval()
        self.register_buffer("lens", lens.to(self._device))
        self.decoder = nn.Sequential(lm.norm, self.model.lm_head)
        self.vocab = word_vocabulary(self.processor.tokenizer)

        # The last kept layer's output lands here on every forward
        self._hidden: Optional[torch.Tensor] = None
        lm.layers[layer].register_forward_hook(self._capture)

    def _capture(self, module: nn.Module, args: tuple[Any, ...], output: Any) -> None:
        """
        Forward hook: keep the hooked decoder layer's output hidden states.
        """
        self._hidden = output[0] if isinstance(output, tuple) else output

    @torch.no_grad()
    def _lens_map(self, image: Image.Image) -> torch.Tensor:
        """
        One image's (D, rows, cols) unit-RMS lens states, float32 on CPU.
        """
        # Chat template with the image and an empty text turn, as upstream
        messages = [{"role": "user", "content": [{"type": "image"}, {"type": "text", "text": ""}]}]
        text = self.processor.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
        inputs = self.processor(text=[text], images=[image], return_tensors="pt")
        inputs = inputs.to(self._device)

        # Truncated forward; the hook keeps the layer output, logits only for the last token
        self.model(**inputs, use_cache=False, logits_to_keep=1)
        image_id = self.processor.tokenizer.convert_tokens_to_ids(self.processor.image_token)
        states = self._hidden[0, inputs["input_ids"][0] == image_id]

        # Unpadded high-res grid: the last rows * (cols + 1) tokens, one newline closing each row
        rows, cols = _anyres_grid(self.processor, inputs)
        grid = states[-(rows * (cols + 1)) :]
        grid = grid.reshape(rows, cols + 1, -1)
        grid = grid[:, :-1]

        # Lens, then unit RMS per state
        lensed = grid.float() @ self.lens.T
        rms = lensed.pow(2).mean(-1, keepdim=True).sqrt()
        lensed = lensed / rms
        lensed = lensed.permute(2, 0, 1)
        return lensed.cpu()

    def forward(self, images: list) -> list[torch.Tensor]:
        """
        Lens states on each image's unpadded AnyRes token grid.

        - one forward per image: AnyRes grids differ in shape
        - the 24x24 overview and row-newline tokens are dropped; the grid spans the full image

        Args:
            images: anything `open_image` accepts, one entry per frame.

        Returns:
            One (D, rows, cols) float32 CPU tensor per image.
        """
        maps = []
        for image in images:

def load_decoder(model_id: str = "llava-hf/llava-v1.6-vicuna-7b-hf", dtype: str = "float16") -> nn.Sequential:
    """
    Final norm + lm_head of a LLaVA-1.6 checkpoint, without loading the rest of the model.

    - reads only the two tensors from the safetensors shards (~260 MB at fp16)
    - hub id or local `save_pretrained` dir; old and new checkpoint key layouts

    Args:
        model_id: hub id or local directory.
        dtype: torch dtype name.

    Returns:
        nn.Sequential(norm, lm_head), eval mode, on the default device.
    """
    config = AutoConfig.from_pretrained(model_id)
    text = config.text_config
    norm = LlamaRMSNorm(text.hidden_size, eps=text.rms_norm_eps)
    lm_head = nn.Linear(text.hidden_size, text.vocab_size, bias=False)

    # Find the two keys: `language_model.model.norm` (hub) or `model.language_model.norm` (new saves)
    index_path = cached_file(model_id, "model.safetensors.index.json")
    weight_map = json.loads(Path(index_path).read_text())["weight_map"]
    norm_key = next(key for key in weight_map if re.search(r"language_model\.(model\.)?norm\.weight$", key))
    head_key = next(key for key in weight_map if key.endswith("lm_head.weight"))

    # Copy each tensor out of its shard
    for module, key in ((norm, norm_key), (lm_head, head_key)):
        shard = cached_file(model_id, weight_map[key])
        with safe_open(shard, framework="pt") as f:
            weight = f.get_tensor(key)
        with torch.no_grad():
            module.weight.copy_(weight)

    decoder = nn.Sequential(norm, lm_head)
    decoder.to(device=get_device(), dtype=getattr(torch, dtype))
    return decoder.eval()
```

```python
def _load_backgrounds() -> list[Image.Image]:
    """
    One mini-imagenet train shard as decoded images (upstream's word-image backgrounds).
    """
    path = hf_hub_download("timm/mini-imagenet", "data/train-00000-of-00013.parquet", repo_type="dataset")
    table = pq.read_table(path, columns=["image"])
    rows = table.column("image").to_pylist()
    return [Image.open(io.BytesIO(row["bytes"])) for row in rows]


def _contrasting_color(patch: Image.Image) -> tuple[int, int, int]:
    """
    Random color at least 90 luminance units from the patch mean; black or white as a fallback.
    """
    pixels = list(patch.getdata())
    lum = sum(0.299 * r + 0.587 * g + 0.114 * b for r, g, b in pixels) / len(pixels)
    for _ in range(20):
        color = (random.randint(0, 255), random.randint(0, 255), random.randint(0, 255))
        if abs(0.299 * color[0] + 0.587 * color[1] + 0.114 * color[2] - lum) > 90:
            return color
    return (0, 0, 0) if lum > 127 else (255, 255, 255)


def _word_image(text: str, backgrounds: list[Image.Image], font: str, size: int = 512) -> Image.Image:
    """
    The word drawn at a random size and position on a random center-cropped background.
    """
    image = random.choice(backgrounds)
    image = image.convert("RGB")
    w, h = image.size
    side = min(w, h)
    image = image.crop(((w - side) // 2, (h - side) // 2, (w + side) // 2, (h + side) // 2))
    image = image.resize((size, size), Image.LANCZOS)

    # Random font size and in-bounds position
    draw = ImageDraw.Draw(image)
    face = ImageFont.truetype(font, random.randint(int(size * 0.10), int(size * 0.22)))
    left, top, right, bottom = draw.textbbox((0, 0), text, font=face)
    tw, th = right - left, bottom - top
    margin = int(size * 0.02)
    x = random.randint(margin - left, max(margin - left, size - tw - margin - left))
    y = random.randint(margin - top, max(margin - top, size - th - margin - top))

    # Color contrasting with the region under the text
    box = (max(0, x + left), max(0, y + top), min(size, x + left + tw), min(size, y + top + th))
    region = image.crop(box)
    draw.text((x, y), text, fill=_contrasting_color(region), font=face)
    return image


@torch.no_grad()
def _head_probs(
    model: LlavaNextForConditionalGeneration,
    processor: LlavaNextProcessor,
    token_ids: list[int],
    images: list[Image.Image],
) -> torch.Tensor:
    """
    Per-head P(word) summed over a batch, (layers, heads).

    - o_proj input at the last position, split per head, through its O slice and the logit lens
    """
    cfg = model.config.text_config
    lm = model.model.language_model
    n_heads = cfg.num_attention_heads
    head_dim = getattr(cfg, "head_dim", None) or cfg.hidden_size // n_heads

    # Transcription prompt with the answer prefilled, one image per row
    messages = [{"role": "user", "content": [{"type": "image"}, {"type": "text", "text": "Transcribe the word in this image."}]}]
    text = processor.apply_chat_template(messages, tokenize=False, add_generation_prompt=True) + "Word:"
    inputs = processor(text=[text] * len(images), images=images, return_tensors="pt")
    inputs = inputs.to(model.device)

    # Capture every layer's o_proj input at the last position
    captured = []
    hooks = [
        layer.self_attn.o_proj.register_forward_pre_hook(lambda module, args: captured.append(args[0][:, -1]))
        for layer in lm.layers
    ]
    try:
        model(**inputs, use_cache=False, logits_to_keep=1)
    finally:
        for hook in hooks:
            hook.remove()

    # Per layer: head outputs -> residual space -> P(target token)
    batch = len(images)
    ids = torch.tensor(token_ids, device=model.device)
    rows = torch.arange(batch, device=model.device)
    out = torch.zeros(len(lm.layers), n_heads, device=model.device)
    for index, layer in enumerate(lm.layers):
        w_o = layer.self_attn.o_proj.weight
        w_o = w_o.view(w_o.shape[0], n_heads, head_dim)
        heads = captured[index].view(batch, n_heads, head_dim)
        projected = torch.einsum("bnh,dnh->bnd", heads, w_o)
        logits = model.lm_head(lm.norm(projected))
        probs = logits.float().softmax(-1)
        out[index] = probs[rows, :, ids].sum(0)
    return out


def score_ocr_heads(
    model_id: str = "llava-hf/llava-v1.6-vicuna-7b-hf",
    *,
    n: int = 1024,
    batch_size: int = 4,
    seed: int = 177,
    dtype: str = "float16",
    font: str = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
) -> torch.Tensor:
    """
    Mean OCR score per attention head: how strongly each head writes the word shown in an image.

    - n single-token vocabulary words, each drawn on a random mini-imagenet background
    - a head's score: P(word) from its O-projected output at the last prompt position
    - loads its own full model: the extractor's is truncated and cannot see heads above `layer`
    - batch_size 16 OOMs on LLaVA-1.6 in 46 GB; run the 7B in tmux

    Args:
        model_id: LLaVA-1.6 hub id or local directory.
        n: number of word images.
        batch_size: images per forward.
        seed: seeds python, numpy and torch RNGs.
        dtype: torch dtype name for the model weights.
        font: TrueType font file the words are drawn in.

    Returns:
        (layers, heads) float32 CPU mean scores; pass as `head_scores`, or `torch.save` and pass the path.
    """
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)

    model = LlavaNextForConditionalGeneration.from_pretrained(model_id, dtype=getattr(torch, dtype))
    model.to(get_device()).eval()
    processor = AutoProcessor.from_pretrained(model_id)

    # Words whose exact lowercase `▁word` token exists, 2-8 letters, as upstream
    vocab = word_vocabulary(processor.tokenizer)
    tokens = processor.tokenizer.convert_ids_to_tokens(vocab.token_ids.tolist())
    pairs = []
    for token, token_id, row in zip(tokens, vocab.token_ids.tolist(), vocab.word_index.tolist()):
        word = token[1:]
        if word == vocab.words[row] and len(word) <= 8:
            pairs.append((word, token_id))
    pairs = random.sample(pairs, k=n)
    backgrounds = _load_backgrounds()

    # Accumulate per-head P(word) over batches
    cfg = model.config.text_config
    total = torch.zeros(cfg.num_hidden_layers, cfg.num_attention_heads, device=model.device)
    for start in tqdm(range(0, n, batch_size), desc="score OCR heads"):
        batch = pairs[start : start + batch_size]
        images = [_word_image(word, backgrounds, font) for word, _ in batch]
        total += _head_probs(model, processor, [token_id for _, token_id in batch], images)

    mean = total / n
    return mean.cpu()
```
