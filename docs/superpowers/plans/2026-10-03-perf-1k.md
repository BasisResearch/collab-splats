```python
def _decode_selection(
    video_path: str,
    chosen: Sequence[int],
    *,
    report: dict,
    on_progress,
    desc: str,
    workers: int = 1,
) -> tuple[list[np.ndarray], list[dict]]:
    """
    Decode exactly the selected frames as RGB, in contiguous ranges on a thread pool.

    - each thread seeks to its range's first frame and keeps only the chosen ones
    - PyAV releases the GIL in decode, so threads scale without pickling frames

    Args:
        video_path: source video.
        chosen: ascending source frame indices.
        report: the quality report the blur_score column is read from.
        on_progress: optional (done, total) callback.
        desc: progress-bar label.
        workers: decode threads; 1 is one linear pass.

    Returns:
        (frames, records) — (H, W, 3) uint8 RGB arrays and their {frame_idx, blur_score} rows.
    """
    laplacian = np.asarray(report["frames"]["laplacian"], dtype=float)

    # Split the selection into one contiguous group per worker
    chosen = list(chosen)
    size = -(-len(chosen) // max(workers, 1))
    groups = [chosen[i : i + size] for i in range(0, len(chosen), size)]

    # Each group decodes its own window and keeps only its chosen frames
    def _decode(group: list[int]) -> dict[int, np.ndarray]:
        """
        Chosen frames of one group, from one seek and one window decode.
        """
        keep = set(group)
        window = iter_frames(video_path, start=group[0], count=group[-1] - group[0] + 1)
        return {idx: bgr for idx, bgr in window if idx in keep}

    decoded: dict[int, np.ndarray] = {}

    with ThreadPoolExecutor(len(groups)) as pool:
        for part in pool.map(_decode, groups):
            decoded.update(part)

    missing = sorted(set(chosen) - decoded.keys())
```

```python
def _unproject_dense(depth: np.ndarray, extrinsics: np.ndarray, intrinsics: np.ndarray, chunk: int = 64) -> np.ndarray:
    """
    World points of every depth pixel, float32 on the GPU in frame chunks.

    - drop-in for vggt's numpy unproject_depth_map_to_point_map
    - depth (k, H, W) or (k, H, W, 1); extrinsics (k, 3, 4) w2c; intrinsics (k, 3, 3)
    - full fp32 matmul: TF32, which a mapanything import enables, would round the rotation
    """
    device = get_device()
    depth = depth.reshape(depth.shape[0], depth.shape[1], depth.shape[2])
    out = np.empty((*depth.shape, 3), dtype=np.float32)
    precision = torch.get_float32_matmul_precision()
    torch.set_float32_matmul_precision("highest")

    # One chunk at a time bounds GPU memory
    try:
        for s in range(0, depth.shape[0], chunk):
            d = torch.from_numpy(depth[s : s + chunk]).to(device, torch.float32)
            e = torch.from_numpy(extrinsics[s : s + chunk]).to(device, torch.float32)
            k = torch.from_numpy(intrinsics[s : s + chunk]).to(device, torch.float32)
            out[s : s + chunk] = unproject(d, e, k).cpu().numpy()
    finally:
        torch.set_float32_matmul_precision(precision)

    return out
```

```python
def test_get_world_grid_matches_numpy_homography():
    """
    GPU float32 world grid equals the per-frame float64 numpy homography to float32 rounding.
    """
    s = _dense_submap(S=3)
    pg = PoseGraph()
    pg.add_submap(s, overlap_frames=1)
    pg.optimize()

    expected = np.empty(s.points.shape, dtype=np.float32)
    for i in range(s.points.shape[0]):
        H = pg.get_homography(s.frame_start + i).astype(np.float64)
```

```python
    def _preprocess(self, paths: list[Path], *, workers: int = 8) -> tuple[Any, np.ndarray]:
        """
        Frames through Omega's crop and resize, plus each frame's crop box.

        - both resize modes first center-crop h / w into [0.5, 2.0]
        - uses the handed-off RGB arrays when every path has one, else reads the files
        - files: upstream on contiguous path chunks in a thread pool; PIL releases the GIL
        - chunks of differing shapes redo one upstream call, which pads them like upstream does
        - returns (N, 3, H, W) images and (N, 6) original_coords
        """
        arrays = None
        if self.frames is not None and all(p.name in self.frames for p in paths):
            arrays = [self.frames[p.name] for p in paths]

        # Each frame's (w, h): from the arrays, else a header read per file
        if arrays is not None:
            sizes = [(a.shape[1], a.shape[0]) for a in arrays]
        else:
            sizes = [Image.open(p).size for p in paths]

        # Compute each frame's aspect-ratio crop box, before any resizing
        rows = []

        for w, h in sizes:
            crop_w, crop_h = w, h

            if h / w < 0.5:
                crop_w = 2 * h
            elif h / w > 2.0:
                crop_h = 2 * w

            rows.append(center_crop_coords((w, h), (w, h), (crop_w, crop_h), (1, 1)))

        original_coords = np.array(rows, dtype=np.float32)

        if arrays is not None:
            return self._load_arrays(arrays, workers=workers), original_coords

        # Split the paths into one contiguous chunk per worker
        files = [str(p) for p in paths]
        size = -(-len(files) // workers)
        chunks = [files[i : i + size] for i in range(0, len(files), size)]

        # Crop and resize each chunk with upstream's own preprocessing
        load = partial(load_and_preprocess_images, image_resolution=self.resolution, mode=self.resize_mode)

        with ThreadPoolExecutor(workers) as pool:
            parts = list(pool.map(load, chunks))

        # Mixed shapes need upstream's list-wide padding
        if len({part.shape[1:] for part in parts}) > 1:
            return load(files), original_coords

        return torch.cat(parts), original_coords

    def _load_arrays(self, arrays: list[np.ndarray], *, workers: int) -> torch.Tensor:
        """
        RGB arrays through upstream's per-image crop, resize and padding, on a thread pool.

        - mirrors load_and_preprocess_images step for step, minus the file open
        - patch size 16, upstream's default
        """
        target = _balanced_target_shape if self.resize_mode == "balanced" else _max_size_target_shape
        to_tensor = TF.ToTensor()

        # Crop, resize and tensorize one frame
        def _one(rgb: np.ndarray) -> torch.Tensor:
            """
            One frame through upstream's crop and bicubic resize.
            """
            image = _crop_to_supported_aspect_ratio(Image.fromarray(rgb))
            width, height = image.size
            target_h, target_w = target(height / max(width, 1), self.resolution, 16)
            return to_tensor(image.resize((target_w, target_h), Image.Resampling.BICUBIC))

        with ThreadPoolExecutor(workers) as pool:
            images = list(pool.map(_one, arrays))

        # Mixed shapes get upstream's padding to a common size
        shapes = {(im.shape[1], im.shape[2]) for im in images}
        if len(shapes) > 1:
            images = _pad_images_to_common_size(images, shapes)

        return torch.stack(images)
```

Design (all exact):
- per window k, the main thread runs:
  1. take forward k's result
  2. GPU post: unproject + retrieval/verify
  3. submit forward k+1 to a 1-worker pool
  4. CPU post: colors, `set_dense_points` percentile, `_add_window_submap` graph solve
- forward k+1 overlaps step 4 only; the order of every op on every window's data is unchanged
- the worker holds `hold_matmul_precision()` for the whole forward; `full_fp32_matmul` takes the same lock
- so no precision flip can land mid-forward: a guarded call on another thread waits instead
- the graph solve is torch-free (`graph.py` has no torch), so it never needs the lock
- `torch.no_grad` is thread-local: the worker function enters it itself
- views are NOT pinned (dropped 2026-10-04 after review: a second ~3.2 GB page-locked host copy, kept alive by `Submap.frames`, for ~3 s)
- MapAnything's list-of-dicts views stay unpinned
- the per-window `torch.cuda.empty_cache()` goes; one stays after the loop
- expected saving, revised: ~15 s CPU overlap + 1.8 s `empty_cache` + ~3 s H2D, so ~20 s, not the spec's 30–35
  - unproject stays serial: it is guarded, and moving it after the launch would make it wait for forward k+1

```python
# Matmul precision is process-wide; one lock orders every change against a running forward
_matmul_precision_lock = threading.RLock()


@contextmanager
def full_fp32_matmul() -> Iterator[None]:
    """
    Run the block at full fp32 matmul precision, then restore the caller's setting.

    - TF32, which a mapanything import enables, would round the matmul
    - the setting is process-wide: waits while another thread holds hold_matmul_precision

    Yields:
        None; the block runs at "highest" precision.
    """
    with _matmul_precision_lock:
        precision = torch.get_float32_matmul_precision()
        torch.set_float32_matmul_precision("highest")

        try:
            yield
        finally:
            torch.set_float32_matmul_precision(precision)


@contextmanager
def hold_matmul_precision() -> Iterator[None]:
    """
    Keep the matmul precision unchanged for the block, across threads.

    - full_fp32_matmul on another thread waits until the block exits
    - wrap a forward that runs while other threads do guarded matmuls

    Yields:
        None.
    """
    with _matmul_precision_lock:
        yield
```

The lock is an `RLock`, so a guarded call nested in the same thread still works. Update the `transform_points` docstring bullet "the setting is process-wide, so the restore is not thread-safe" to "the toggle waits while another thread holds hold_matmul_precision".
